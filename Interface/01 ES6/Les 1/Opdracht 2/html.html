const multiply = (a, b) => {
    return a * b;
};

const greet = (name) => {
    return `Hello, ` + name + `!`;
};

const double = (num) => {
    return num * 2;
};

const filterEvens = (arr) => arr.filter(num => num % 2 === 0);

console.log(multiply(3, 4));
console.log(greet("Alice"));
console.log(double(5));
console.log(filterEvens([1, 2, 3, 4, 5, 6]));

let globalVar = 10;

let myObject = {
    waarde: 5,
    regularFunction: function() {
        console.log(this.waarde);
        console.log(globalVar);
    },

    arrowFunction: () => {
        console.log(this.waarde);
        console.log(globalVar);
    }
};

myObject.regularFunction();
myObject.arrowFunction();

// regular = 5, 10
// arrow = undefined, 10
// ze kunnen allebij globalvar zien maar de arrow functie kan de waarde van 'this' niet zien omdat arrow functies geen 'this' hebben
